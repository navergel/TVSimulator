LG 32LV340C 
  Connection: Baud:9600bps Data:8bits Parity:None  StopBit:1bit CommunicationCode:ASCII code  *Use a crossed (reversed) cable
  
SetId = 1-99 if if value is 0, controls every connected set, in hex
[Cr] = 0x0D
[ ]  = 0x20
[command1][command2][ ][SetId][ ][Data][Cr]
[OK] = 0x4F 0x4B   //this is for Ack
[NG] = 0x4E 0x47   //this is for Ack
[x]  = 0x78
[k]  = 0x6B
[a]  = 0x61
[b]  = 0x62
[e]  = 0x65
[f]  = 0x66
[F]  = 0x46
[0]  = 0x30
[1]  = 0x31
